# 스마트 컨트랙트 보안 취약점과 감사 방법

## 개요

스마트 컨트랙트는 한 번 배포되면 수정이 불가능하고, 코드 자체가 곧 법이 되는 특성을 갖는다. 이 불변성(Immutability)은 블록체인의 핵심 강점이지만, 동시에 보안 취약점이 존재할 경우 치명적인 결과를 초래한다. 2016년 The DAO 해킹 사건(약 6천만 달러 손실), 2022년 Ronin Bridge 해킹(약 6억 달러 손실) 등은 스마트 컨트랙트 보안이 단순한 코드 품질 문제가 아니라 실질적인 금전적 위험임을 보여준다.

이 글에서는 실무에서 반드시 알아야 할 스마트 컨트랙트의 주요 취약점 유형과, 이를 탐지하고 예방하기 위한 감사(Audit) 방법론을 다룬다. Solidity 예제 코드와 함께 각 취약점의 동작 방식, 방어 패턴, 자동화 도구까지 실무 관점에서 정리한다.

## 핵심 개념: 주요 취약점 유형

### 1. 재진입 공격 (Reentrancy Attack)

재진입 공격은 외부 컨트랙트 호출 시 상태 변수가 업데이트되기 전에 공격자가 함수를 재귀적으로 호출하는 방식이다. The DAO 해킹의 원인이 된 취약점이기도 하다.

**취약한 코드 예시:**

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract VulnerableBank {
    mapping(address => uint256) public balances;

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    // 취약점: 상태 업데이트 전에 외부 호출 발생
    function withdraw() external {
        uint256 amount = balances[msg.sender];
        require(amount > 0, "No balance");

        // 1. 외부 호출 먼저 발생
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "Transfer failed");

        // 2. 상태 업데이트가 나중에 발생 -> 재진입 허용
        balances[msg.sender] = 0;
    }
}
```

**공격자 컨트랙트:**

```solidity
contract Attacker {
    VulnerableBank public target;

    constructor(address _target) {
        target = VulnerableBank(_target);
    }

    function attack() external payable {
        target.deposit{value: msg.value}();
        target.withdraw();
    }

    // fallback에서 재귀 호출
    receive() external payable {
        if (address(target).balance >= 1 ether) {
            target.withdraw();
        }
    }
}
```

**방어 패턴 (Checks-Effects-Interactions):**

```solidity
contract SecureBank {
    mapping(address => uint256) public balances;
    // ReentrancyGuard 패턴
    bool private locked;

    modifier noReentrant() {
        require(!locked, "Reentrant call");
        locked = true;
        _;
        locked = false;
    }

    function withdraw() external noReentrant {
        uint256 amount = balances[msg.sender];
        require(amount > 0, "No balance");

        // Checks -> Effects -> Interactions 순서 준수
        balances[msg.sender] = 0;                        // Effects 먼저
        (bool success, ) = msg.sender.call{value: amount}(""); // 마지막에 Interactions
        require(success, "Transfer failed");
    }
}
```

### 2. 정수 오버플로우/언더플로우 (Integer Overflow/Underflow)

Solidity 0.8.0 이전 버전에서는 정수 연산 시 오버플로우/언더플로우가 자동으로 처리되지 않아 예상치 못한 값이 발생할 수 있었다.

```solidity
// Solidity < 0.8.0 에서의 취약한 예시
pragma solidity ^0.7.0;

contract VulnerableToken {
    mapping(address => uint256) public balances;

    function transfer(address to, uint256 amount) external {
        // balances[msg.sender]가 0일 때 amount를 빼면 언더플로우 발생
        // 예: 0 - 1 = 2^256 - 1 (매우 큰 값)
        balances[msg.sender] -= amount;
        balances[to] += amount;
    }
}
```

**방어 패턴:**

```solidity
// Solidity >= 0.8.0: 기본적으로 오버플로우 체크 내장
pragma solidity ^0.8.0;

// 또는 OpenZeppelin SafeMath 사용 (< 0.8.0 환경)
import "@openzeppelin/contracts/utils/math/SafeMath.sol";

contract SecureToken {
    using SafeMath for uint256;
    mapping(address => uint256) public balances;

    function transfer(address to, uint256 amount) external {
        balances[msg.sender] = balances[msg.sender].sub(amount, "Insufficient balance");
        balances[to] = balances[to].add(amount);
    }
}
```

### 3. 접근 제어 취약점 (Access Control)

온체인에서 권한 관리를 잘못 설정하면 누구나 관리자 기능을 호출할 수 있게 된다.

```solidity
// 취약한 예시: 초기화 함수에 접근 제어 없음
contract VulnerableProxy {
    address public owner;
    bool public initialized;

    // 누구나 호출 가능한 초기화 함수
    function initialize(address _owner) external {
        // initialized 체크 없음
        owner = _owner;
    }
}

// 안전한 예시: OpenZeppelin Ownable 활용
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/proxy/utils/Initializable.sol";

contract SecureProxy is Initializable, Ownable {
    function initialize(address _owner) external initializer {
        _transferOwnership(_owner);
    }

    function sensitiveOperation() external onlyOwner {
        // 오너만 호출 가능
    }
}
```

### 4. 프론트런닝 (Front-Running)

블록체인의 투명성으로 인해 멤풀(mempool)에 있는 트랜잭션을 감지하고, 더 높은 가스비를 제시해 먼저 실행될 수 있다.

```solidity
// 취약한 경매 컨트랙트
contract VulnerableAuction {
    address public highestBidder;
    uint256 public highestBid;

    function bid() external payable {
        require(msg.value > highestBid, "Bid too low");
        highestBidder = msg.sender;
        highestBid = msg.value;
    }
}

// 방어 패턴: Commit-Reveal 스킴
contract SecureAuction {
    struct Commit {
        bytes32 commitment;
        bool revealed;
    }

    mapping(address => Commit) public commits;
    mapping(address => uint256) public bids;
    uint256 public commitDeadline;
    uint256 public revealDeadline;

    // 1단계: 비드 금액을 해시로 커밋
    function commit(bytes32 _commitment) external {
        require(block.timestamp < commitDeadline, "Commit phase over");
        commits[msg.sender] = Commit(_commitment, false);
    }

    // 2단계: 실제 금액과 솔트 공개
    function reveal(uint256 _amount, bytes32 _salt) external payable {
        require(block.timestamp >= commitDeadline && block.timestamp < revealDeadline);
        require(!commits[msg.sender].revealed, "Already revealed");

        bytes32 commitment = keccak256(abi.encodePacked(_amount, _salt, msg.sender));
        require(commitment == commits[msg.sender].commitment, "Invalid reveal");

        commits[msg.sender].revealed = true;
        bids[msg.sender] = _amount;
    }
}
```

## 실전 예제: 스마트 컨트랙트 감사 자동화

### Slither를 활용한 정적 분석

[Slither](https://github.com/crytic/slither)는 Trail of Bits가 개발한 Solidity 정적 분석 도구로, 다양한 취약점 패턴을 자동으로 탐지한다.

```bash
# 설치
pip3 install slither-analyzer

# 단일 파일 분석
slither contracts/MyContract.sol

# Hardhat/Foundry 프로젝트 분석
slither . --hardhat-artifacts-directory artifacts

# 특정 탐지기만 실행
slither . --detect reentrancy-eth,unprotected-upgrade

# JSON 리포트 출력
slither . --json slither-report.json

# 심각도 필터링 (high만 표시)
slither . --filter-paths "node_modules" --exclude-informational --exclude-low
```

### Mythril을 활용한 심볼릭 실행 분석

```bash
# 설치
pip3 install mythril

# 기본 분석
myth analyze contracts/MyContract.sol

# 바이트코드 분석
myth analyze --bin-runtime <bytecode>

# 최대 탐색 깊이 설정 (복잡한 컨트랙트)
myth analyze contracts/MyContract.sol --max-depth 15 --solver-timeout 10000

# JSON 출력
myth analyze contracts/MyContract.sol -o json
```

### Foundry를 활용한 Fuzzing 테스트

```solidity
// test/FuzzTest.t.sol
pragma solidity ^0.8.0;

import "forge-std/Test.sol";
import "../src/SecureBank.sol";

contract SecureBankFuzzTest is Test {
    SecureBank bank;
    address alice = address(0x1);

    function setUp() public {
        bank = new SecureBank();
        vm.deal(alice, 100 ether);
    }

    // Foundry가 amount 값을 랜덤으로 변경하며 테스트
    function testFuzz_depositAndWithdraw(uint96 amount) public {
        vm.assume(amount > 0 && amount <= 10 ether);

        vm.startPrank(alice);
        bank.deposit{value: amount}();

        uint256 balanceBefore = address(alice).balance;
        bank.withdraw();
        uint256 balanceAfter = address(alice).balance;

        // 잔액이 정확히 복구되어야 함
        assertEq(balanceAfter - balanceBefore, amount);
        assertEq(bank.balances(alice), 0);
        vm.stopPrank();
    }

    // 불변식(Invariant) 테스트
    function invariant_totalBalanceMatchesContractBalance() public {
        // 컨트랙트 ETH 잔액은 항상 전체 사용자 잔액 합과 같아야 함
        assertEq(address(bank).balance, bank.totalDeposits());
    }
}
```

```bash
# Foundry 퍼징 실행
forge test --match-test testFuzz_ -v

# Invariant 테스트
forge test --match-test invariant_ --invariant-runs 1000
```

## 주의사항 및 트레이드오프

### 보안 vs 가스 비용

`ReentrancyGuard`, `Pausable` 같은 보안 패턴은 추가적인 스토리지 읽기/쓰기를 발생시켜 가스 비용을 증가시킨다. 특히 L1 이더리움에서는 이 비용이 체감될 수 있다. 따라서 위험도가 낮은 읽기 전용 함수에는 불필요한 가드를 추가하지 않는 것이 좋다.

### 업그레이드 가능성 vs 불변성

Proxy 패턴(UUPS, Transparent Proxy)을 사용하면 버그 수정이 가능하지만, 업그레이드 권한 자체가 새로운 공격 벡터가 된다. 반드시 **TimeLock**과 **멀티시그(Multisig)** 를 함께 적용해야 한다.

```solidity
// TimeLock을 통한 업그레이드 지연 예시
import "@openzeppelin/contracts/governance/TimelockController.sol";

// 최소 48시간 후에 실행 가능
uint256 minDelay = 48 hours;
address[] memory proposers = [teamMultisig];
address[] memory executors = [teamMultisig];

TimelockController timelock = new TimelockController(minDelay, proposers, executors, address(0));
```

### 외부 오라클 의존성

Chainlink 등 외부 오라클에서 가격 데이터를 가져올 때 단일 소스에만 의존하면 오라클 조작 공격에 취약해진다. TWAP(Time-Weighted Average Price)을 사용하거나 복수 오라클을 활용해 검증해야 한다.

### 자동화 도구의 한계

Slither, Mythril 같은 자동화 도구는 **알려진 패턴**의 취약점만 탐지한다. 비즈니스 로직 오류, 경제적 설계 결함(토크노믹스 취약점), 복잡한 프로토콜 간 상호작용 취약점은 **수동 코드 리뷰**와 **전문 감사 기관의 감사**를 통해서만 발견할 수 있다. 자동화 도구는 보완재이지 대체재가 아니다.

실무에서 적용할 수 있는 간략한 감사 프로세스:

| 단계 | 내용 | 도구 |
|------|------|------|
| 정적 분석 | 코드 패턴 기반 취약점 탐지 | Slither, Semgrep |

| 테스트 커버리지 | 분기/라인 커버리지 100% 목표 | Foundry Coverage, Hardhat |
| 외부 감사 | 제3자 전문 기관 감사 | Trail of Bits, OpenZeppelin, CertiK |

## 정리

스마트 컨트랙트 보안은 단일 도구나 체크리스트로 완결되지 않는다. 재진입 공격, 정수 오버플로우, 접근 제어, 프론트런닝 등 각 취약점 유형을 깊이 이해하고, **Checks-Effects-Interactions 패턴**, **ReentrancyGuard**, **OpenZeppelin 검증 라이브러리** 활용을 코딩 표준으로 정착시켜야 한다.

자동화 분석 도구(Slither, Mythril, Foundry Fuzzing)는 개발 파이프라인에 CI/CD로 통합해 지속적으로 실행하고, 메인넷 배포 전에는 반드시 전문 감사 기관의 외부 감사를 받아야 한다. 특히 TVL(Total Value Locked)이 높은 DeFi 프로토콜이라면 버그 바운티 프로그램(Immunefi 등)을 병행 운영하는 것이 표준적인 접근법이다.

보안은 배포 전 한 번의 이벤트가 아니라, 설계부터 운영까지 이어지는 지속적인 프로세스임을 항상 염두에 두자.
